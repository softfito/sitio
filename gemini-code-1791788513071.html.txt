    <script>
        function copiarClabe() {
            const clabe = document.getElementById('clabeText').innerText.trim();
            const btn = document.getElementById('copyBtn');

            navigator.clipboard.writeText(clabe).then(() => {
                btn.innerText = "¡CLABE Copiada con Éxito! ✓";
                btn.style.backgroundColor = "#0284c7";
                
                setTimeout(() => {
                    btn.innerText = "Copiar CLABE al Portapapeles";
                    btn.style.backgroundColor = "#16a34a";
                }, 3000);
            }).catch(err => {
                alert('No se pudo copiar automáticamente. Por favor selecciónala manualmente.');
            });
        }
